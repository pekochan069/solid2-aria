## Usage

```tsx showLineNumbers
import {
  getLocalTimeZone,
  today,
  type CalendarDate,
} from "@internationalized/date";

import { Calendar } from "#components/ui/calendar";
```

```tsx showLineNumbers
import { createSignal } from "solid-js";

const [date, setDate] = createSignal<CalendarDate | null>(
  today(getLocalTimeZone())
);

return <Calendar value={date()} onChange={setDate} class="rounded-lg border" />;
```

## Date Picker

You can use the `<Calendar>` component to build a date picker. Compose `Calendar` with `Popover`, `PopoverTrigger`, and a button that displays the selected date.

## Custom Cell Size

<ComponentPreview
  styleName="aria-nova"
  name="calendar-custom-days"
  title="Custom Cell Size"
  description="A calendar with custom cell size that's responsive."
  class="**:[.preview]:h-[560px]"
/>

You can customize the size of calendar cells using the `--cell-size` CSS variable. You can also make it responsive by using breakpoint-specific values:

```tsx showLineNumbers
<Calendar
  value={date}
  onChange={setDate}
  class="rounded-lg border [--cell-size:--spacing(11)] md:[--cell-size:--spacing(12)]"
/>
```

Or use fixed values:

```tsx showLineNumbers
<Calendar
  value={date}
  onChange={setDate}
  class="rounded-lg border [--cell-size:2.75rem] md:[--cell-size:3rem]"
/>
```
